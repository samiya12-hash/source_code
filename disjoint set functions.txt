ll find(ll a,ll parent[])
{
   vi v;
   while(parent[a]>0)
   {
       v.pb(a);
       a = parent[a];
   }
   for(ll i=0;i<v.size();i++)parent[v[i]] = a;
   return a;
}



void unio(ll a,ll b,ll parent[])//parent array
{
    parent[a] += parent[b];
    parent[b] = a;
}
